## 📝 页面对象实现

基于以上分析，页面对象实现如下：

```python
# pages/common/login_page.py

class LoginPage(BasePage):
    """登录页面 - 基于实际页面分析（2026-03-24）"""
    
    # ===== 元素定位器（已验证）=====
    USERNAME_INPUT = "input[placeholder*='用户名']"
    PASSWORD_INPUT = "input[placeholder*='密码']"
    VERIFY_CODE_INPUT = "input[placeholder*='验证码']"
    VERIFY_CODE_IMAGE = "img[src*='verifyCode']"
    LOGIN_BUTTON = "button:has-text('登录')"
    FORGOT_PASSWORD_LINK = "a:has-text('忘记密码')"
    ERROR_MESSAGE = ".error-message"  # 待验证
    
    # ===== 页面方法 =====
    def login(self, username: str, password: str, verify_code: str = "2222"):
        """执行登录"""
        logger.info(f"登录：用户={username}")
        
        (self
         .fill(self.USERNAME_INPUT, username, "用户名")
         .fill(self.PASSWORD_INPUT, password, "密码")
         .fill(self.VERIFY_CODE_INPUT, verify_code, "验证码")
         .click(self.LOGIN_BUTTON, "登录按钮"))
        
        return self
    
    def is_logged_in(self) -> bool:
        """判断是否登录成功"""
        current_url = self.page.url
        return "index#" in current_url or "myarchive" in current_url
    
    def get_error_message(self) -> str:
        """获取错误消息"""
        try:
            return self.page.locator(self.ERROR_MESSAGE).inner_text()
        except:
            return ""
```
